## 常用命令 (本地)

```powershell
# 后端 API
cd d:\ZY\code\backend
py -3.10 -m uvicorn app.main:app --host 127.0.0.1 --port 8000

# RAG 服务
cd d:\ZY\code\rag
py -3.10 server.py

# 解密命例库 (加密文件时)
py -3.10 d:\ZY\file_decrypt.py "d:\ZY\数据库\01八字命理\C_命例经验"

# 重抽命例
cd d:\ZY\code\knowledge\scripts
py -3.10 extract_cases.py

# 重建 01 八字 RAG (带 textRole)
cd d:\ZY\code\rag
py -3.10 -c "import chromadb; from chromadb.config import Settings; from config import CHROMA_DIR; c=chromadb.PersistentClient(path=str(CHROMA_DIR), settings=Settings(anonymized_telemetry=False)); c.delete_collection('kb_01_bazi')"
py -3.10 build_index.py --no-reset --category 01八字命理

# 测试
cd d:\ZY\code\backend
py -3.10 -m pytest tests/test_case_match.py tests/test_arbitrator_conflicts.py -q
```
